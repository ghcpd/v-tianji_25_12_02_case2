### 7. **Code Migration Examples**

#### **Example 1: React 18 Root API**
```javascript
// BEFORE (index.js)
import ReactDOM from 'react-dom';
ReactDOM.render(
  <Provider store={store}>
    <BrowserRouter><App /></BrowserRouter>
  </Provider>,
  document.getElementById('root')
);

// AFTER
import { createRoot } from 'react-dom/client';
const root = createRoot(document.getElementById('root'));
root.render(
  <Provider store={store}>
    <BrowserRouter><App /></BrowserRouter>
  </Provider>
);
```

#### **Example 2: React Router 6 Migration**
```javascript
// BEFORE (App.js)
import { Route, Switch, withRouter } from 'react-router-dom';

<Switch>
  <Route exact path="/" component={Dashboard} />
  <Route path="/products" component={Products} />
</Switch>

export default withRouter(connect(mapStateToProps)(App));

// AFTER
import { Routes, Route, useLocation } from 'react-router-dom';

<Routes>
  <Route path="/" element={<Dashboard />} />
  <Route path="/products" element={<Products />} />
</Routes>

export default connect(mapStateToProps)(App);
// Use useLocation() hook inside component
```

#### **Example 4: Moment → date-fns**
```javascript
// BEFORE
import moment from 'moment';
const formattedDate = moment().format('MMMM Do YYYY, h:mm:ss a');

// AFTER
import { format, differenceInDays } from 'date-fns';
const formattedDate = format(new Date(), 'MMMM do yyyy, h:mm:ss a');
const daysAgo = differenceInDays(new Date(), new Date(item.date));
```
